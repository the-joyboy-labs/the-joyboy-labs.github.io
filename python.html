// FEATURED TOOL

A preview-first downloads organizer

Sort common file types into folders. The script starts in preview mode, skips name collisions, and only moves files after you opt in.

TOOL 01 / FILE MANAGEMENT

Organize a local Downloads folder

Review the preview output first. Change DRY_RUN to False only when you are ready to move files.

PYTHON
organize_downloads.py
from pathlib import Path
import shutil

DOWNLOADS = Path.home() / "Downloads"
CATEGORIES = {
    ".png": "Images",
    ".jpg": "Images",
    ".pdf": "Documents",
    ".zip": "Archives",
}
DRY_RUN = True

def organize_folder(folder: Path = DOWNLOADS) -> None:
    for item in folder.iterdir():
        if not item.is_file():
            continue

        category = CATEGORIES.get(item.suffix.lower())
        if category is None:
            continue

        target_dir = folder / category
        destination = target_dir / item.name
        if destination.exists():
            print(f"skip {item.name}: destination already exists")
            continue

        action = "preview" if DRY_RUN else "move"
        print(f"{action}: {item.name} -> {category}/")
        if not DRY_RUN:
            target_dir.mkdir(exist_ok=True)
            shutil.move(str(item), str(destination))


if __name__ == "__main__":
    organize_folder()
